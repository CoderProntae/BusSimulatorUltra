name: Build APK

on:
  push:
    branches: [ dev, main, 'arena/**' ]
  workflow_dispatch:

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - name: Download assets (CI has full internet)
        run: |
          set +e
          mkdir -p res/assets/textures/asphalt res/assets/textures/concrete res/assets/textures/wall res/assets/environment res/assets/models
          wget -L -q -O asphalt.zip "https://ambientcg.com/get?file=Asphalt030_1K-JPG.zip" && unzip -o -j asphalt.zip "*Color*.jpg" -d res/assets/textures/asphalt/ && unzip -o -j asphalt.zip "*Normal*.jpg" -d res/assets/textures/asphalt/ && unzip -o -j asphalt.zip "*Roughness*.jpg" -d res/assets/textures/asphalt/
          wget -q -O concrete.zip "https://dl.polyhaven.org/file/ph-assets/Textures/zip/1k/concrete_floor_02.zip" && unzip -o -j concrete.zip "*.jpg" -d res/assets/textures/concrete/
          wget -q -O wall.zip "https://dl.polyhaven.org/file/ph-assets/Textures/zip/1k/brick_wall_001.zip" && unzip -o -j wall.zip "*.jpg" -d res/assets/textures/wall/
          wget -q -O sky_download.hdr "https://dl.polyhaven.org/file/ph-assets/HDRIs/hdr/1k/kloofendal_48d_partly_cloudy_puresky_1k.hdr" && test -s sky_download.hdr && mv sky_download.hdr res/assets/environment/sky.hdr
          wget -q -O bus_download.glb "https://raw.githubusercontent.com/KhronosGroup/glTF-Sample-Assets/main/Models/CesiumMilkTruck/glTF-Binary/CesiumMilkTruck.glb" && test -s bus_download.glb && mv bus_download.glb res/assets/models/bus.glb || echo "bus fallback CSG"
          (cd res/assets/textures/asphalt && for f in *Color*; do mv "$f" albedo.jpg; done; for f in *Normal*; do mv "$f" normal.jpg; done; for f in *Roughness*; do mv "$f" roughness.jpg; done) || true
          (cd res/assets/textures/concrete && for f in *.jpg; do case "$f" in *[Cc]olor*|*albedo*) mv "$f" albedo.jpg ;; *[Nn]ormal*|*normal*) mv "$f" normal.jpg ;; esac; done) || true
          (cd res/assets/textures/wall && for f in *.jpg; do case "$f" in *[Cc]olor*|*albedo*) mv "$f" albedo.jpg ;; *[Nn]ormal*|*normal*) mv "$f" normal.jpg ;; esac; done) || true
          echo "---- downloaded assets ----"; find res/assets -type f -exec ls -la {} \;

      - name: Verify required asset paths exist (procedural fallbacks are committed)
        run: |
          set -e
          MISSING=0
          for f in res/assets/textures/asphalt/albedo.jpg res/assets/textures/asphalt/normal.jpg res/assets/textures/asphalt/roughness.jpg res/assets/textures/concrete/albedo.jpg res/assets/textures/concrete/normal.jpg res/assets/textures/wall/albedo.jpg res/assets/textures/wall/normal.jpg res/assets/environment/sky.hdr; do
            if [ -s "$f" ]; then
              echo "OK   $f"
            else
              echo "MISS $f"
              MISSING=1
            fi
          done
          if [ "$MISSING" = "1" ]; then
            echo "A required texture is missing and had no committed fallback."
            exit 1
          fi

      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: 17

      - uses: android-actions/setup-android@v3

      - name: Download Godot
        run: |
          wget -q https://github.com/godotengine/godot/releases/download/4.4-stable/Godot_v4.4-stable_linux.x86_64.zip
          unzip -o Godot_v4.4-stable_linux.x86_64.zip
          mv Godot_v4.4-stable_linux.x86_64 godot && chmod +x godot

      - name: Templates
        run: |
          wget -q https://github.com/godotengine/godot/releases/download/4.4-stable/Godot_v4.4-stable_export_templates.tpz
          mkdir -p ~/.local/share/godot/export_templates/4.4.stable
          unzip -o Godot_v4.4-stable_export_templates.tpz
          mv templates/* ~/.local/share/godot/export_templates/4.4.stable/

      - name: Keystore
        run: keytool -genkeypair -v -keystore debug.keystore -alias androiddebugkey -keyalg RSA -keysize 2048 -validity 10000 -storepass android -keypass android -dname "CN=Debug,O=Debug,C=US"

      - name: Patch export preset
        run: |
          KEYSTORE_PATH=$(pwd)/debug.keystore
          sed -i "s|custom_template/debug=\".*\"|custom_template/debug=\"\"|" res/export_presets.cfg
          sed -i "s|custom_template/release=\".*\"|custom_template/release=\"\"|" res/export_presets.cfg
          sed -i "s|keystore/debug=\".*\"|keystore/debug=\"$KEYSTORE_PATH\"|" res/export_presets.cfg
          sed -i "s|keystore/debug_user=\".*\"|keystore/debug_user=\"androiddebugkey\"|" res/export_presets.cfg
          sed -i "s|keystore/debug_password=\".*\"|keystore/debug_password=\"android\"|" res/export_presets.cfg
          sed -i "s|keystore/release=\".*\"|keystore/release=\"\"|" res/export_presets.cfg
          sed -i "s|keystore/release_user=\".*\"|keystore/release_user=\"\"|" res/export_presets.cfg
          sed -i "s|keystore/release_password=\".*\"|keystore/release_password=\"\"|" res/export_presets.cfg
          echo "---- export_presets.cfg ----"
          cat res/export_presets.cfg

      - name: Import
        run: ./godot --headless --path res --import --quit || true

      - name: Install GDScript parser
        run: pip install "gdtoolkit==4.*"

      - name: Check every GDScript parses
        run: |
          set -e
          gdparse res/scripts/*.gd
          echo "all GDScript files parsed"

      - name: Check for inline-if inside % format tuples
        run: |
          set -e
          if grep -nE "%[[:space:]]*[(\"']" res/scripts/*.gd | grep -E "\bif\b.*\belse\b"; then
            echo "ERROR: inline-if inside a % format tuple (breaks the GDScript parser)"
            exit 1
          fi
          echo "no %-tuple ternaries"

      # Informational only (continue-on-error): a false alarm from the dummy
      # headless renderer must never block the APK. Real script errors are
      # still printed loudly in the log and marked as a failed step.
      - name: Smoke test (boots the real project, autoloads registered)
        continue-on-error: true
        run: |
          set +e
          # NOTE: do NOT use "--check-only --script", it loads each file in
          # isolation where the GameState autoload is not registered yet and
          # reports a false "Identifier not found: GameState" compile error.
          # Booting the project for a few frames validates the scripts the way
          # the game actually runs them.
          ./godot --headless --path res --quit-after 180 2>&1 | tee smoke.log
          echo "---- scanning for real script errors ----"
          # Ignored on purpose (harmless headless/dummy-renderer noise):
          #   "Do not use progress dialog (task)", "task_step ... canceled",
          #   "Parameter \"t\" is null" (dummy texture storage),
          #   "cannot connect to daemon at tcp:5037" (ADB, no device in CI)
          if grep -E "SCRIPT ERROR|Parse Error|Compile Error" smoke.log; then
            echo "FAILED: a script error occurred while booting the project"
            exit 1
          fi
          echo "project booted with no script errors"

      - name: Export
        run: |
          mkdir -p build
          ./godot --headless --path res --export-debug "Android" "$(pwd)/build/BusSimulator.apk"
          ls -la build/

      - name: APK info
        run: |
          ls -la build/BusSimulator.apk
          du -h build/BusSimulator.apk

      - uses: actions/upload-artifact@v4
        if: always()
        with:
          name: BusSimulator-APK
          path: build/BusSimulator.apk
          retention-days: 30
          if-no-files-found: error
